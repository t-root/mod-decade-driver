#include <Arduino.h>

#define S1_PIN 1
#define S2_PIN 3

void setup() {
  Serial.begin(115200);
  delay(1000);

  pinMode(S1_PIN, INPUT_PULLUP);
  pinMode(S2_PIN, INPUT_PULLUP);

  Serial.println("s1\ts2");
}

void loop() {
  int s1 = !digitalRead(S1_PIN); // đảo lại
  int s2 = !digitalRead(S2_PIN); // đảo lại

  Serial.print(s1);
  Serial.print("\t");
  Serial.println(s2);

  delay(200);
}
